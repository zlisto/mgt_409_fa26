The Scenario

Build a stock research chatbot. People should be able to open a web app, chat with the bot (“What’s Apple’s recent volatility?” / “Any news on Microsoft?”), and get answers grounded in live market data and web search results.

The Task

1. Create an AGENTS.md with basic rules you’ll reuse all semester: which AI model to use, how to handle API keys, and the app look you want (for example a modern black-and-pink style). Also include Dash run rules: turn off the auto-reloader (or run with debug off) so one edit doesn’t spawn many servers, and when stopping the app, kill the whole process — not only whatever is on port 8050.

2. Paste the app.py chat shell from below into your vibe coder. Ask it to restyle that UI to your taste (colors, layout, fonts — keep the chat mechanics: message history, a text box, Enter / Send to submit, how replies appear, thinking dots). When it looks right, save what you learned as a reusable chatbot skill you’ll use for chat apps all semester.

3. Use that skill (and this shell) to finish the stock research chat app. Fill in get_reply() so the model can call tools:

  • Live market data from Yahoo Finance (never invent prices)
  • Custom tools for financial metrics (price, momentum, volatility, Sharpe, and similar)
  • A web search tool for recent financial news

The model should return live answers and show you when a tool ran.

def render_messages(history, pending):
    blocks = []
    for msg in history or []:
        who = "Bot" if msg["role"] == "assistant" else "You"
        blocks.append(
            html.Div(
                [html.Strong(f"{who}: "), dcc.Markdown(msg["content"])],
                style={"marginBottom": "10px"},
            )
        )
        tools = msg.get("tool_events") or []
        if tools:
            blocks.append(
                html.Ul(
                    [html.Li(f"Used: {t.get('name', '?')}") for t in tools],
                    className="chat-tool-steps",
                )
            )
    if pending:
        blocks.append(
            html.Div(
                [html.Strong("Bot "), html.Span(className="chat-thinking")],
                style={"color": "#888"},
                role="status",
                **{"aria-live": "polite"},
            )
        )
    return blocks


@app.callback(Output("chat-window", "children"), Input("chat-history", "data"), Input("pending-request", "data"))
def update_chat(history, pending):
    return render_messages(history, pending)


@app.callback(
    Output("chat-history", "data", allow_duplicate=True),
    Output("pending-request", "data"),

def submit_message(_, text, history, pending):
    if pending or not text or not text.strip():
        return no_update, no_update, no_update, no_update, no_update, no_update
    updated = [*(history or []), {"role": "user", "content": text.strip()}]
    return updated, {"text": text.strip()}, "", False, 0, 1


@app.callback(
    Output("chat-history", "data", allow_duplicate=True),
    Output("pending-request", "data", allow_duplicate=True),
    Output("agent-poll", "disabled", allow_duplicate=True),
    Input("agent-poll", "n_intervals"),
    State("pending-request", "data"),
    State("chat-history", "data"),
    prevent_initial_call=True,
)
def complete_chat_turn(_, pending, history):
    if not pending:
        return no_update, no_update, True
    try:
        print(f"
chat turn: {pending['text']!r}", flush=True)
        result = get_reply(pending["text"], (history or [])[:-1])
        tools = result.get("tool_events", []) if isinstance(result, dict) else []
        text = result["text"] if isinstance(result, dict) else str(result)
        if tools:
            print("tools this turn: " + ", ".join(t.get("name", "?") for t in tools), flush=True)
        else:
            print("tools this turn: (none)", flush=True)
        reply = {"role": "assistant", "content": text, "tool_events": tools}
    except Exception:
        traceback.print_exc()
        reply = {"role": "assistant", "content": "Something went wrong. Check .env / Portkey and try again."}
    return [*(history or []), reply], None, True
